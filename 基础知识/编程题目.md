# 1.岛屿数量
```C++
class Solution {
  int n, m;
  int dtx[4] = {-1, 0,1,0}, dty[4] = {0,-1,0,1};
public:
  int numIsLands(vector<vector(char>>& grid) {
      int ans = 0;
      m = grid.size(), n = grid[0].size();
      for (int i = 0; i < m; i++){
        for (int j = 0; j < n; j++) {
            if (grid[i][j] == '1') {
              dfs(i,j,grid)
              ans++:
            }
        }
    }
      return ans;
  }

void dfs(int x, int y, vector<vector(char>>& grid)
{
  grid[x][y] = '0';
  for (int i = 0; i < 4; i++) {
    x += dtx[i], y += dty[i];
    if (x >= 0 && x < n && y >= 0 && y < m && grid[x][y] == '1') {
       dfs(x,y,grid);
    }
  }
}
};
```
