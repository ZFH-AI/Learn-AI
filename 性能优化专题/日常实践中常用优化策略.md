/**
 * 文件功能：将常见的优化方式汇总保存
 * 文件作者：ZFH/z00465944
 * 
 * 性能优化的目标：
 *  1、时间性能，减小系统执行时间
 *  2、空间性能，减少系统占用的空间
 * 
 * 性能优化的策略：
 *  1、分解：系统包含多个子系统或子模块，分别查看其执行路径、函数、指令等
 *  2、测量：在分解的基础上测量耗时的整体情况以及耗时分布，先抓主要矛盾后处理次要矛盾
 * 
 * X核性能优化的差异：以CacheMiss为例来说明
 *  【单核】
 *     1、CacheMiss的原因一般是: 1):代码路径太长超出Cache的容量，Cache需要替换，以至于CacheMiss
 *                              2):判断条件较多，路径选择较多，执行路径相互交替，Cache需要替换
 *                              3):I\D Cache相互替换[对I-Cache的优化一般方式：
 *                                                  1）减少无用或冗余代码 
 *                                                  2）减少函数调用开销 
 *                                                  3）关键路径代码短小精悍，相关代码紧邻]
 *    2、D-Cache Miss的原因一般是：1):数据结构太大，超出Cache容量
 *                                2):访问的字段分散需要多次读取
 *   【多核】
 *      1、多核独享L1 Cache，共享L3 Cache，L2既可以独享也可以共享看硬件实现，这就造成当存在多条执行路径时Cache冲突就会增大
 *      2、D-Cache Miss：锁、原子操作、伪共享否会引起多核上的CacheMiss
 * 
 * 高效编码
 *  【变量与访存优化建议】
 *     1):初始化构造：关键路径、流程中需要频繁构建的静态数据可以考虑在初始化时构造，在路径中复用
 *     2):内存初始化：在定义数据结构之后如果memset之后立马有对其进行赋值，建议去掉memset
 *     3):结构体赋值尽量按定义顺序
 *     4):尽可能使用const常量
 *     5):减少全局变量的访问
 *     6):局部变量尽量减少大堆栈、大内存的分配
 *     7):数据结构按字节对齐
 * 
 *   【循环优化建议】

// 1、循环展开、降低循环的层数或次数
void LoopOptim() {
    int i = 0;
    int count  = 10000;
    int a[10001];
    // 优化前
    while (i < count) {
        a[i] = i;
        i++;
    }
    // 优化后,这里循环展开1次，其实可以循环展开多次
    while (i < count - 1) {
        a[i] = i;
        a[i + 1] = i + 1;
        i += 2;
    }
}

// 3、循环内计算（每次计算为不变值或定值）外提，降低无效计算
void LoopOptim3() 
{
    // 优化前
    for (int i = 0; i < GetMaxNum(); i++) {
        // DO-WORK;
    }
    // 优化后
    int num = GetMaxNum();
    for (int i = 0; i < num; i++) {
        // DO-WORK
    }

    // 优化前
    int a[1000];
    int num1, num2, num3;
    for (int i = 0; i < 1000; i++) {
        a[i] = (i + 1) * num1 * num2 * num3;
    }
    // 优化后
    int total = num1 * num2 * num3;
    int value = total;
    for (int i = 0; i < 1000; i++) {
        a[i] = value;
        value += total;
    }
}

// 6、多重循环，将循环次数最多的放在循环最内部
void LoopOptim6()
{
    // 优化前
    for (int i = 0; i < 100; i++) {
        for (int j = 0; j < 5; j++) {
            // DO-Work;
        }
    }

    // 优化后
    for (int j = 0; j < 5; j++) {
        for (int i = 0; i < 100; i++) {
            // DO-Work;
        }
    }
    // 优化前：100 + 100*5 = 600次
    // 优化后：5 + 5 *100  = 505次
}
