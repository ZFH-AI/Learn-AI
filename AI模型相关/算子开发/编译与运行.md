typedef int8_t S8X16 __attribute__((__vector_size__(16)));
typedef int8_t S8X8 __attribute__((__vector_size__(8)));
__attribute__((optimize("unroll-loops")))
void fooSIMD(int16_t value[32], int16_t value_2[32], int16_t active[32], uint32_t usrNum)
{
    S16X8* v_active = (S16X8*) active;
    S16X8* v_old = (S16X8*) value_2;
    S16X8* v_new = (S16X8*) value;

    for (UINT32 usrLoopNum = 0; usrLoopNum < 4; usrLoopNum++) {
        v_old[usrLoopNum] = ((v_new[usrLoopNum] << 10)  + (((v_old[usrLoopNum] - v_new[usrLoopNum] )) << 4) * v_active[usrLoopNum])>>10;
        v_active[usrLoopNum] = (S16X8){0} + 1;
    }
}
